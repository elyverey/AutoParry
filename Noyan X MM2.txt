local firstTab = true
local function AddTab(enName, trName)
    local Page = Instance.new("ScrollingFrame")
    Page.Size = UDim2.new(1, 0, 1, 0)
    Page.Position = UDim2.new(0, 0, 0, 0)
    Page.BackgroundTransparency = 1
    Page.BorderSizePixel = 0
    Page.ScrollBarThickness = 4
    Page.CanvasSize = UDim2.new(0, 0, 0, 0)
    Page.Visible = firstTab
    Page.Parent = PagesFolder

    local PageList = Instance.new("UIListLayout", Page)
    PageList.Padding = UDim.new(0, 8)
    PageList.HorizontalAlignment = Enum.HorizontalAlignment.Center
    Instance.new("UIPadding", Page).PaddingTop = UDim.new(0, 12)

    PageList:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
        Page.CanvasSize = UDim2.new(0, 0, 0, PageList.AbsoluteContentSize.Y + 25)
    end)

    local TabBtn = Instance.new("TextButton")
    TabBtn.Size = UDim2.new(0.92, 0, 0, 32)
    TabBtn.BackgroundColor3 = Color3.fromRGB(22, 22, 30)
    TabBtn.Text = string.format(" 📎  %s", enName)
    TabBtn.TextColor3 = firstTab and ThemeColor or Color3.fromRGB(180, 180, 200)
    TabBtn.Font = Enum.Font.GothamSemibold
    TabBtn.TextSize = 11
    TabBtn.TextXAlignment = Enum.TextXAlignment.Left
    TabBtn.Parent = Sidebar
    Instance.new("UICorner", TabBtn).CornerRadius = UDim.new(0, 6)

    table.insert(C.TranslatedElements, {
        Type = "Tab",
        Button = TabBtn,
        EN = enName,
        TR = trName
    })

    TabBtn.MouseButton1Click:Connect(function()
        for _, p in pairs(PagesFolder:GetChildren()) do p.Visible = false end
        for _, b in pairs(Sidebar:GetChildren()) do
            if b:IsA("TextButton") then b.TextColor3 = Color3.fromRGB(180, 180, 200) end
        end
        Page.Visible = true
        TabBtn.TextColor3 = ThemeColor
    end)
    
    firstTab = false

    local Tab = {}
    function Tab:AddToggle(enTitle, trTitle, default, callback)
        local state = default
        local TBtn = Instance.new("TextButton")
        TBtn.Size = UDim2.new(0.92, 0, 0, 36)
        TBtn.BackgroundColor3 = PanelBg
        TBtn.Text = "  " .. enTitle
        TBtn.TextColor3 = state and ThemeColor or Color3.fromRGB(200, 200, 220)
        TBtn.Font = Enum.Font.GothamSemibold
        TBtn.TextSize = 11
        TBtn.TextXAlignment = Enum.TextXAlignment.Left
        TBtn.Parent = Page
        Instance.new("UICorner", TBtn).CornerRadius = UDim.new(0, 6)
        
        local Stroke = Instance.new("UIStroke", TBtn)
        Stroke.Color = state and ThemeColor or Color3.fromRGB(40, 40, 55)

        table.insert(C.TranslatedElements, {
            Type = "Element",
            Button = TBtn,
            EN = "  " .. enTitle,
            TR = "  " .. trTitle
        })

        TBtn.MouseButton1Click:Connect(function()
            state = not state
            TBtn.TextColor3 = state and ThemeColor or Color3.fromRGB(200, 200, 220)
            Stroke.Color = state and ThemeColor or Color3.fromRGB(40, 40, 55)
            pcall(callback, state)
        end)
    end

    function Tab:AddButton(enTitle, trTitle, callback)
        local BBtn = Instance.new("TextButton")
        BBtn.Size = UDim2.new(0.92, 0, 0, 36)
        BBtn.BackgroundColor3 = PanelBg
        BBtn.Text = "  " .. enTitle
        BBtn.TextColor3 = Color3.fromRGB(255, 200, 100)
        BBtn.Font = Enum.Font.GothamSemibold
        BBtn.TextSize = 11
        BBtn.TextXAlignment = Enum.TextXAlignment.Left
        BBtn.Parent = Page
        Instance.new("UICorner", BBtn).CornerRadius = UDim.new(0, 6)
        Instance.new("UIStroke", BBtn).Color = Color3.fromRGB(50, 50, 70)

        table.insert(C.TranslatedElements, {
            Type = "Element",
            Button = BBtn,
            EN = "  " .. enTitle,
            TR = "  " .. trTitle
        })

        BBtn.MouseButton1Click:Connect(function()
            pcall(callback)
        end)
    end

    return Tab
end

local function executeInstantFling(targetChar)
    pcall(function()
        local localChar = LocalPlayer.Character
        local localHrp = localChar and localChar:FindFirstChild("HumanoidRootPart")
        local targetHrp = targetChar and targetChar:FindFirstChild("HumanoidRootPart")
        local localHum = localChar and localChar:FindFirstChild("Humanoid")
        
        if localHrp and targetHrp and localHum then
            local originalPos = localHrp.CFrame
            
            for _, part in pairs(localChar:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.CustomPhysicalProperties = PhysicalProperties.new(100, 0, 0, 100, 100)
                end
            end

            local connection
            connection = RunService.Heartbeat:Connect(function()
                pcall(function()
                    if targetHrp and localHrp then
                        localHrp.CFrame = targetHrp.CFrame * CFrame.new(0, 0, 0)
                        localHrp.AssemblyLinearVelocity = Vector3.new(30000, 30000, 30000)
                        localHrp.AssemblyAngularVelocity = Vector3.new(99999, 99999, 99999)
                        targetHrp.AssemblyLinearVelocity = Vector3.new(math.random(-10000, 10000), 100000, math.random(-10000, 10000))
                    end
                end)
            end)

            task.wait(0.25)
            if connection then connection:Disconnect() end
            
            localHrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            localHrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            localHrp.CFrame = originalPos
        end
    end)
end
