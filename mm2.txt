-- Helper Functions
local function getPlayerRole(player)
    local character = player.Character
    if not character then return "Innocent" end
    local tool = character:FindFirstChildWhichIsA("Tool")
    if tool then
        if tool.Name:lower():find("knife") or tool.Name:lower():find("murder") then
            return "Murderer"
        elseif tool.Name:lower():find("gun") or tool.Name:lower():find("pistol") then
            return "Sheriff"
        end
    end
    return "Innocent"
end

local function getAlivePlayers()
    local alive = {}
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("Humanoid") and player.Character.Humanoid.Health > 0 then
            table.insert(alive, player)
        end
    end
    return alive
end

local function getMurderer()
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and getPlayerRole(player) == "Murderer" then
            return player
        end
    end
    return nil
end

local function getClosestPlayer()
    local alive = getAlivePlayers()
    if #alive == 0 then return nil end
    local closest = nil
    local dist = math.huge
    local pos = RootPart.Position
    for _, player in pairs(alive) do
        local char = player.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            local d = (char.HumanoidRootPart.Position - pos).Magnitude
            if d < dist then
                dist = d
                closest = player
            end
        end
    end
    return closest
end

local function getPlayerFromCharacter(char)
    for _, player in pairs(Players:GetPlayers()) do
        if player.Character == char then
            return player
        end
    end
    return nil
end

local function cleanupFly()
    if flyConnection then
        flyConnection:Disconnect()
        flyConnection = nil
    end
    if flyBodyGyro then
        flyBodyGyro:Destroy()
        flyBodyGyro = nil
    end
    if flyBodyVelocity then
        flyBodyVelocity:Destroy()
        flyBodyVelocity = nil
    end
end

-- Fly System
local function toggleFly(state)
    flyEnabled = state
    if state then
        cleanupFly()
        flyBodyGyro = Instance.new("BodyGyro")
        flyBodyVelocity = Instance.new("BodyVelocity")
        flyBodyGyro.P = 9e4
        flyBodyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
        flyBodyGyro.CFrame = RootPart.CFrame
        flyBodyVelocity.MaxForce = Vector3.new(9e9, 9e9, 9e9)
        flyBodyVelocity.Velocity = Vector3.new(0, 0, 0)
        flyBodyGyro.Parent = RootPart
        flyBodyVelocity.Parent = RootPart
        
        flyConnection = RunService.Heartbeat:Connect(function()
            if not flyEnabled then
                cleanupFly()
                return
            end
            local moveDirection = Vector3.new(0, 0, 0)
            if flyKeys.W then moveDirection = moveDirection + RootPart.CFrame.LookVector end
            if flyKeys.S then moveDirection = moveDirection - RootPart.CFrame.LookVector end
            if flyKeys.A then moveDirection = moveDirection - RootPart.CFrame.RightVector end
            if flyKeys.D then moveDirection = moveDirection + RootPart.CFrame.RightVector end
            if flyKeys.Space then moveDirection = moveDirection + Vector3.new(0, 1, 0) end
            if flyKeys.Shift then moveDirection = moveDirection - Vector3.new(0, 1, 0) end
            
            if moveDirection.Magnitude > 0 then
                moveDirection = moveDirection.Unit * flySpeed
            end
            if flyBodyVelocity then
                flyBodyVelocity.Velocity = moveDirection
            end
            if flyBodyGyro then
                flyBodyGyro.CFrame = RootPart.CFrame
            end
        end)
    else
        cleanupFly()
    end
end

-- ESP System
local function createESP(player)
    local character = player.Character
    if not character then return end
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    
    local role = getPlayerRole(player)
    local color = Color3.fromRGB(255, 255, 255)
    if role == "Murderer" then
        color = Color3.fromRGB(255, 0, 0)
    elseif role == "Sheriff" then
        color = Color3.fromRGB(0, 100, 255)
    else
        color = Color3.fromRGB(0, 255, 0)
    end
    
    local box = Instance.new("BoxHandleAdornment")
    box.Size = Vector3.new(4, 6, 2)
    box.Adornee = hrp
    box.Color3 = color
    box.Transparency = 0.5
    box.AlwaysOnTop = true
    box.ZIndex = 0
    box.Parent = hrp
    
    local line = Instance.new("SelectionPartLasso")
    line.Part = hrp
    line.Color = color
    line.Transparency = 0.7
    line.AlwaysOnTop = true
    line.Parent = character
    
    espBoxes[player] = box
    espLines[player] = line
end

local function updateESP()
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            createESP(player)
        end
    end
end

local function clearESP()
    for _, box in pairs(espBoxes) do
        pcall(function() box:Destroy() end)
    end
    for _, line in pairs(espLines) do
        pcall(function() line:Destroy() end)
    end
    espBoxes = {}
    espLines = {}
end

local function toggleESP(state)
    espEnabled = state
    if state then
        clearESP()
        updateESP()
        if espConnection then espConnection:Disconnect() end
        espConnection = RunService.Heartbeat:Connect(function()
            if not espEnabled then
                espConnection:Disconnect()
                espConnection = nil
                clearESP()
                return
            end
            clearESP()
            updateESP()
        end)
    else
        if espConnection then
            espConnection:Disconnect()
            espConnection = nil
        end
        clearESP()
    end
end

-- Auto Shoot (Sheriff)
local function toggleAutoShoot(state)
    autoShootEnabled = state
    if state then
        if shootConnection then shootConnection:Disconnect() end
        shootConnection = RunService.Heartbeat:Connect(function()
            if not autoShootEnabled then
                shootConnection:Disconnect()
                shootConnection = nil
                return
            end
            
            local murderer = getMurderer()
            if not murderer then
                toggleAutoShoot(false)
                WindUI:Notify({
                    Title = "Auto Shoot",
                    Content = "Murderer not found, disabling",
                    Icon = "x-circle",
                    Duration = 2
                })
                return
            end
            
            local mChar = murderer.Character
            if not mChar or not mChar:FindFirstChild("HumanoidRootPart") or not mChar:FindFirstChild("Humanoid") or mChar.Humanoid.Health <= 0 then
                toggleAutoShoot(false)
                WindUI:Notify({
                    Title = "Auto Shoot",
                    Content = "Murderer killed, disabling",
                    Icon = "check-circle",
                    Duration = 2
                })
                return
            end
            
            local origin = RootPart.Position + Vector3.new(0, 1.5, 0)
            local targetPos = mChar.HumanoidRootPart.Position + Vector3.new(0, 1.5, 0)
            local direction = (targetPos - origin).Unit * 500
            local ray = Ray.new(origin, direction)
            local hit, position = Workspace:FindPartOnRayWithIgnoreList(ray, {Character, mChar})
            
            if hit and hit:IsDescendantOf(mChar) then
                local dist = (RootPart.Position - mChar.HumanoidRootPart.Position).Magnitude
                if dist < 100 then
                    if tick() - lastShootTime >= shootCooldown then
                        lastShootTime = tick()
                        local tool = Character:FindFirstChildWhichIsA("Tool")
                        if tool and tool:FindFirstChild("Handle") then
                            tool:Activate()
                            task.wait(math.random(1, 5) / 100)
                        end
                    end
                end
            end
        end)
    else
        if shootConnection then
            shootConnection:Disconnect()
            shootConnection = nil
        end
    end
end

-- Aimbot
local function toggleAimbot(state)
    aimbotEnabled = state
    if state then
        if aimbotConnection then aimbotConnection:Disconnect() end
        aimbotConnection = RunService.Heartbeat:Connect(function()
            if not aimbotEnabled then
                aimbotConnection:Disconnect()
                aimbotConnection = nil
                return
            end
            local murderer = getMurderer()
            if not murderer or not murderer.Character or not murderer.Character:FindFirstChild("HumanoidRootPart") then
                return
            end
            if murderer == LocalPlayer then return end
            local targetPos = murderer.Character.HumanoidRootPart.Position
            local currentPos = RootPart.Position
            local direction = (targetPos - currentPos).Unit
            local newCFrame = CFrame.lookAt(currentPos, currentPos + direction)
            RootPart.CFrame = RootPart.CFrame:Lerp(newCFrame, aimbotSpeed / 100)
        end)
    else
        if aimbotConnection then
            aimbotConnection:Disconnect()
            aimbotConnection = nil
        end
    end
end

-- Coin Farm
local function toggleCoinFarm(state)
    coinFarmEnabled = state
    if state and not isFarming then
        isFarming = true
        coroutine.wrap(function()
            while coinFarmEnabled do
                local coins = {}
                for _, v in pairs(Workspace:GetDescendants()) do
                    if v:IsA("BasePart") and v.Name:lower():find("coin") then
                        table.insert(coins, v)
                    end
                end
                if #coins > 0 then
                    local noclipConnection
                    noclipConnection = RunService.Stepped:Connect(function()
                        if not coinFarmEnabled then
                            noclipConnection:Disconnect()
                            return
                        end
                        for _, part in pairs(Character:GetDescendants()) do
                            if part:IsA("BasePart") then
                                part.CanCollide = false
                            end
                        end
                    end)
                    
                    for _, coin in pairs(coins) do
                        if not coinFarmEnabled then break end
                        local tweenInfo = TweenInfo.new(
                            1 / coinFarmSpeed,
                            Enum.EasingStyle.Linear,
                            Enum.EasingDirection.InOut
                        )
                        local tween = TweenService:Create(RootPart, tweenInfo, {CFrame = coin.CFrame * CFrame.new(0, 2, 0)})
                        tween:Play()
                        tween.Completed:Wait()
                        if ReplicatedStorage:FindFirstChild("CoinCollectionEvent") then
                            pcall(function()
                                ReplicatedStorage.CoinCollectionEvent:FireServer(coin)
                            end)
                        end
                        task.wait(0.1)
                    end
                    
                    noclipConnection:Disconnect()
                    for _, part in pairs(Character:GetDescendants()) do
                        if part:IsA("BasePart") then
                            part.CanCollide = true
                        end
                    end
                else
                    task.wait(1)
                end
            end
            isFarming = false
        end)()
    else
        isFarming = false
        coinFarmEnabled = false
        for _, part in pairs(Character:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = true
            end
        end
    end
end

local function playSound()
    local sound = Instance.new("Sound")
    sound.SoundId = "rbxassetid://9120383714"
    sound.Volume = 1
    sound.Parent = Workspace
    sound:Play()
    task.wait(sound.TimeLength or 1)
    sound:Destroy()
end

local function triggerKillEffect(position)
    if not killEffectEnabled then return end
    if killEffectType == "Explosion" then
        createExplosion(position)
    elseif killEffectType == "Flash" then
        createFlash(position)
    elseif killEffectType == "Sound" then
        playSound()
    elseif killEffectType == "All" then
        createExplosion(position)
        createFlash(position)
        playSound()
    end
end

local function setupKillEffect()
    Humanoid.Died:Connect(function()
        if killEffectEnabled then
            triggerKillEffect(RootPart.Position)
        end
    end)
end
